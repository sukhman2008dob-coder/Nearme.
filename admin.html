Optional: Enable Multi-Device Cloud Traffic Logging

Right now, the Admin Panel tracks bookings and listings from Supabase, and logs client events automatically. To also store visitor traffic across all user devices permanently in Supabase, run this short SQL query in your Supabase SQL Editor: 

-- 1. Create Analytics table for live traffic and events
create table if not exists site_analytics (
    id bigint generated by default as identity primary key,
    event_type text not null,
    session_id text,
    device text,
    page_path text,
    referrer text,
    event_data jsonb default '{}'::jsonb,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- 2. Allow public inserts from visitor browsers
alter table site_analytics enable row level security;
create policy "Allow anonymous inserts" on site_analytics
    for insert with check (true);
create policy "Allow admin read" on site_analytics
    for select using (true);
